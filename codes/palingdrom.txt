# Function to compute GCD
gcd() {
  local x=$1
  local y=$2
  while [ $y -ne 0 ]; do
    local temp=$y
    y=$((x % y))
    x=$temp
  done
  echo $x
}

# Function to compute LCM
lcm() {
  local x=$1
  local y=$2
  echo $((x * y / $(gcd $x $y)))
}

# Compute LCM of a & b
lcm_ab=$(lcm $a $b)
echo "LCM of $a and $b is: $lcm_ab"

# Compute GCD of b & c

echo "GCD of $b and $c is: $gcd_bc"
